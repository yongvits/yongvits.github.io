// State let currentFile = null; let isConverting = false; let ffmpegLoaded = false; // FFmpeg Instances let ffmpeg = null; let fetchFile = null; // Custom Toast Notification System function showToast(message, type = 'info') { setTimeout(() => toast.remove(), 3000); } async function loadFFmpeg() { if (ffmpegLoaded) return; try { showToast('Initializing FFmpeg engine... (Downloading core files)', 'info'); // Ensure CDN scripts are fully loaded before destructuring if (!window.FFmpegWASM || !window.FFmpegUtil) { throw new Error("FFmpeg libraries not loaded. Check internet connection or CDN."); } const { FFmpeg } = window.FFmpegWASM; const { fetchFile: _fetchFile, toBlobURL } = window.FFmpegUtil; ffmpeg = new FFmpeg(); fetchFile = _fetchFile; ffmpeg.on('progress', ({ progress, time }) => { const percent = Math.round(progress * 100); // Prevent showing NaN or infinity if(percent >= 0 && percent <= 100) { progressText.textContent = `Converting: ${percent}%`; progressBar.style.width = `${percent}%`; } }); // Load single-thread core using jsDelivr for better reliability in Asia const baseURL = 'https://cdn.jsdelivr.net/npm/@ffmpeg/core@0.12.6/dist/umd'; await ffmpeg.load({ coreURL: await toBlobURL(`${baseURL}/ffmpeg-core.js`, 'text/javascript'), wasmURL: await toBlobURL(`${baseURL}/ffmpeg-core.wasm`, 'application/wasm'), }); ffmpegLoaded = true; } catch (error) { console.error('Error loading FFmpeg:', error); showToast('Load Error: ' + error.message, 'error'); resetUI(); throw error; // Stop execution if load fails } } // Event Listeners
